select o.name,ddps.row_count
from sys.indexes as i
inner join sys.objects as o on i.object_id=o.object_id
inner join sys.dm_db_partition_stats as ddps on i.object_id = ddps.object_id
and i.index_id = ddps.index_id
where i.index_id<2 and o.is_ms_shipped = 0
order by ddps.row_count desc

with duplicaterecords as (
			                                    SELECT id,  name , row_number() OVER (PARTITION BY name ORDER BY id) as ranknumber  
			                                    FROM products),
												finalrecords
												as
												(
												select firstrows.id firstid, source.id,source.name from duplicaterecords source
												inner join duplicaterecords firstrows on firstrows.name = source.name and firstrows.ranknumber = 1
												where source.ranknumber>1
												)

update products set isactive = false from finalrecords 
where products.id = finalrecords.id;


update lrinvoiceitems set productid = firstid from finalrecords 
where lrinvoiceitems.productid = finalrecords.id;

update stock set productid = firstid from finalrecords 
where stock.productid = finalrecords.id;
